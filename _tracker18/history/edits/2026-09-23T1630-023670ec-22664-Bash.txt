// Ramy, 23 Sep 2026, asked for this and in the same breath: "just make sure

/** Two columns out of Sheets or Excel: group then candidate, tab separated. */
function parseRosterPaste(text) {
  var rows = [], problems = [];
  String(text || '').split(/\r?\n/).forEach(function (line, i) {
    if (!line.trim()) return;
    // Sheets and Excel both put TABS on the clipboard. Commas and semicolons
    // are accepted too, for a CSV saved out and reopened.
    var cells = line.split(/\t|,|;/).map(function (c) { return c.replace(/^["\s]+|["\s]+$/g, ''); }).filter(function (c) { return c !== ''; });
    if (!cells.length) return;
    // A header row, however it is worded.
    if (i === 0 && /^(group|level|class)$/i.test(cells[0]) && /^(candidate|name|trainee)$/i.test(cells[1] || '')) return;
    if (cells.length < 2) { problems.push('Line ' + (i + 1) + ' has only "' + cells[0] + '" -- each line needs a group and a name'); return; }
    rows.push({ group: cells[0], name: cells.slice(1).join(' ') });
  });
  var seen = {};
  rows.forEach(function (r) {
    var k = slug(r.group + '-' + r.name);
    if (seen[k]) problems.push('Two candidates would share one record: ' + r.group + ' / ' + r.name);
    seen[k] = true;
  });
  return { rows: rows, problems: problems };
}

    var names = Object.keys(groups);
    out.innerHTML = parsed.rows.length
