new = """// Everything the record says about one candidate, read together: the teaching
// practice, the four assignments and the Stage 2 review. Ramy, 12 Sep 2026 --
// "it should be an indication of both TPs, assignments, and stage two."
//
// A candidate is judged on where they stand NOW, not on their worst day: it is
// the LATEST graded TP that decides, and an earlier not-to-standard becomes a
// watch rather than a risk once the next one is to standard. That is how a
// course tutor actually reads a profile, and it is what stops week one from
// condemning somebody for the rest of the course.
function readCandidate(c){
var risk = [], watch = [], strong = false;

var latest = null, latestN = 0, graded = 0, notStd = 0, above = 0, earlyNotStd = [];
for (var k = 1; k <= 8; k++) {
var g = tpGrade(c['tp'+k]);
if (!g) continue;
graded++; latest = g; latestN = k;
if (g.key === 'NOTSTD') { notStd++; earlyNotStd.push(k); }
if (g.key === 'ABOVE') above++;
}
if (latest && latest.key === 'NOTSTD') risk.push('TP'+latestN+' not to standard');
else if (notStd > 0) watch.push(notStd === 1 ? 'TP'+earlyNotStd[0]+' was not to standard' : notStd+' TPs not to standard');
if (graded > 0 && above === graded) strong = true;

var fails = [], resubs = [];
ASSIGNMENTS.forEach(function(code){
var v = c[code.toLowerCase()+'_r'] || '';
if (v === 'FAIL') fails.push(code);
else if (v === 'RES') resubs.push(code);
});
if (fails.length) risk.push(fails.join(' and ') + (fails.length > 1 ? ' failed' : ' failed'));
if (resubs.length >= 2) watch.push(resubs.length + ' assignments passed on resubmission');

if (c.stage2 === 'NOTSTD') risk.push('Stage 2 not to standard');
if (c.stage2 === 'ABOVE') strong = true;

var recorded = graded > 0 || fails.length > 0 || resubs.length > 0 || !!c.stage2;
var state = (c.withdrawn === true || c.withdrawn === 'true') ? 'withdrawn'
: risk.length ? 'risk'
: watch.length ? 'watch'
: !recorded ? 'none'
: strong ? 'above' : 'ontrack';
return { state: state, risk: risk, watch: watch, graded: graded, notStd: notStd, fails: fails, resubs: resubs };
}

var STATE_CHIP = {
withdrawn: { text:'WITHDRAWN', bg:'#EEF2F5', color:'var(--grey)' },
risk:      { text:'NEEDS ATTENTION', bg:'#FBEAE9', color:'var(--rose)' },

ontrack:   { text:'ON TRACK', bg:'#E4F0F8', color:'var(--teal)' },
none:      { text:'NOT STARTED', bg:'#EEF2F5', color:'var(--grey)' }
};
function statusFor(c){ return STATE_CHIP[readCandidate(c).state]; }"""
