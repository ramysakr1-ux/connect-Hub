import io, re
s = io.open("index.html", encoding="utf-8").read()
# lift the real functions out of the page, not a copy of them
for name in ["slug", "parseRosterPaste"]:
    m = re.search(r"\nfunction " + name + r"\s*\(", s)
    i = m.start() + 1
    j = s.index("{", i); k = j; d = 0
    while True:
        if s[k] == "{": d += 1
        elif s[k] == "}":
            d -= 1
            if d == 0: break
        k += 1
    print(s[i:k+1])
