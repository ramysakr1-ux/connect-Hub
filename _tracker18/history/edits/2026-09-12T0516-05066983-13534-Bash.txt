new_fn = """function readCandidate(c){
var latest = null, latestN = 0, graded = 0, notStd = 0, above = 0, lateNotStd = [];
for (var k = 1; k <= 8; k++) {
var g = tpGrade(c['tp'+k]);
if (!g) continue;
graded++; latest = g; latestN = k;
if (g.key === 'NOTSTD') { notStd++; if (k >= SECOND_HALF_FROM) lateNotStd.push(k); }
if (g.key === 'ABOVE') above++;
}

// Where they stand FOR THIS STAGE of the course. A tutorial is the formal
// judgement and the later one is the more current, so Stage 3 outranks
// Stage 2, which outranks any single lesson; between tutorials the latest
// TP is the best evidence there is.
var standard = null, from = '';
if (c.stage3) { standard = c.stage3; from = 'Stage 3'; }
else if (c.stage2) { standard = c.stage2; from = 'Stage 2'; }
else if (latest) { standard = latest.key; from = 'TP' + latestN; }

// The written assignments cap the GRADE; they do not change the standard.
// Handbook eligibility: fail more than one and a Pass is not available at
// all; fail exactly one and a Pass may still be recommended, but not Pass A.
var fails = [], resubs = [];
ASSIGNMENTS.forEach(function(code){
var v = c[code.toLowerCase()+'_r'] || '';
if (v === 'FAIL') fails.push(code);
else if (v === 'RES') resubs.push(code);
});
var ceiling = fails.length > 1
? { level: 'nopass', text: fails.join(', ') + ' failed — not eligible for a Pass' }
: fails.length === 1
? { level: 'nopassa', text: fails[0] + ' failed — a Pass is still open, but not Pass A' }
: null;

// CELTA 5, Stage Three Progress Record: the record "must be completed by
// tutors in the final third of the course" for candidates who a) were not
// to standard at Stage 2, b) were at standard at Stage 2 but are not making
// the expected progress in the second half, or c) were above standard at
// Stage 2 and likewise. Stage 2 is the trigger, so nothing is claimed until
// a Stage 2 verdict exists, and a recorded Stage 3 answers it.
var stage3Why = null;
if (c.stage2 && !c.stage3) {
if (c.stage2 === 'NOTSTD') stage3Why = 'not to standard at Stage 2';
else if (lateNotStd.length) {
stage3Why = (c.stage2 === 'ABOVE' ? 'above standard' : 'to standard') + ' at Stage 2, then TP' +
lateNotStd.join(' and TP') + ' not to standard';
}
}

var why = [];
if (standard === 'NOTSTD') why.push(from + ' not to standard');
var recorded = graded > 0 || fails.length > 0 || resubs.length > 0 || !!c.stage2 || !!c.stage3;
var state = (c.withdrawn === true || c.withdrawn === 'true') ? 'withdrawn'
: !recorded ? 'none'
: standard === 'NOTSTD' ? 'notstd'
: standard === 'ABOVE' ? 'above'
: standard === 'STD' ? 'std'
: 'none';
return {
state: state, from: from, why: why, graded: graded, notStd: notStd,
fails: fails, resubs: resubs, ceiling: ceiling, stage3Why: stage3Why
};
}

"""
